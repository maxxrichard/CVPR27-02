\section{Proofs}
\label{sec:proofs}
Throughout, $Q_\lambda = \alpha I + \lambda L_T$, $N_z = \diag(n)$, and for a fixed feature dimension $j$ we write $d = d_{:j}$, $e = e_{:j}$ and $s = s_j$.

\paragraph{Derivation of \cref{eq:dsolve}.}
Given $z$, the terms of \cref{eq:objective} that depend on $D$ are $\sum_{i,k} z_{ik}\frac{(x_{ij} - t_{kj} - d_{kj})^2}{2\kappa\sigma_j^2}$ summed over $j$, plus $\frac{1}{2\kappa\bar\sigma^2}\sum_j d_{:j}^{\top}Q_\lambda d_{:j}$. Expanding the square around $\bar{x}_{kj}$ and dropping terms that do not depend on $d$ gives, per dimension, $\frac{1}{2\kappa\sigma_j^2}\sum_k n_k (d_k - e_k)^2 + \frac{1}{2\kappa\bar\sigma^2} d^{\top}Q_\lambda d$. Multiplying by $\kappa\sigma_j^2$ and setting the gradient to zero yields $(N_z + s_j Q_\lambda)d = N_z e$, which is \cref{eq:dsolve}. The matrix is positive definite whenever $\alpha>0$ or $n_k>0$ for some $k$ in every connected component, so the solution is unique.

\paragraph{Proof of \cref{prop:special}.}
With $\lambda=0$ the system is diagonal: $(n_k + s\alpha)d_k = n_k e_k$. With $\alpha = 0$ and $L_T$ the normalised Laplacian of the complete graph, $L_T = \frac{K}{K-1}(I - \frac{1}{K}\mathbf{1}\mathbf{1}^{\top})$, whose null space is spanned by $\mathbf{1}$. As $\lambda\to\infty$ the minimiser of $\frac12\sum_k n_k(d_k-e_k)^2 + \frac{\lambda s}{2}d^{\top}L_Td$ is forced into the null space, $d = \delta\mathbf{1}$, and minimising over $\delta$ gives $\delta = \sum_k n_k e_k/\sum_k n_k$. \qed

\paragraph{Proof of \cref{prop:absent}.}
Partition the system by $A$ and $P$. Rows in $A$ have $n_k = 0$ and $e_k = 0$, so they read $s\,(Q_{\lambda,AA}d_A + Q_{\lambda,AP}d_P) = 0$, which gives the harmonic extension $d_A = -Q_{\lambda,AA}^{-1}Q_{\lambda,AP}d_P$ ($Q_{\lambda,AA}$ is positive definite when $\alpha>0$, or when every component of $A$ touches $P$). Substituting into the $P$ rows gives $(N_{z,P} + s(Q_{\lambda,PP} - Q_{\lambda,PA}Q_{\lambda,AA}^{-1}Q_{\lambda,AP}))d_P = N_{z,P}e_P$, which is \cref{eq:dsolve} on $P$ with the Schur complement $Q_\lambda/Q_{\lambda,AA}$, the Kron reduction of the graph onto $P$~\cite{dorfler2013kron}. \qed

\paragraph{Proof of \cref{prop:variance}.}
Given $z$, the per-dimension posterior of $d$ is Gaussian with precision $(N_z + sQ_\lambda)/(\kappa\sigma_j^2)$, so its covariance is $\kappa\sigma_j^2(N_z + sQ_\lambda)^{-1}$. For symmetric positive definite $A$ and positive semi-definite $B$, $(A+B)^{-1}\preceq A^{-1}$, hence the diagonal entries satisfy $[(N_z + s\alpha I + s\lambda L_T)^{-1}]_{kk}\le [(N_z + s\alpha I)^{-1}]_{kk} = 1/(n_k + s\alpha)$. Strictness: $[(A+B)^{-1}]_{kk} = [A^{-1}]_{kk}$ for all $k$ in a set $C$ would require $B$ to vanish on the corresponding block of the Woodbury expansion, which fails for any class $k$ with $(L_T)_{kk}>0$, that is, any class with at least one neighbour. \qed

\paragraph{Proof of \cref{prop:monotone}.}
With $\lambda_L = 0$ the objective is, in $z$, a sum of $N$ strictly convex functions on the simplex whose minimisers are the softmax in \cref{eq:estep} (standard mean-field argument), in $\pi$ a Dirichlet-regularised multinomial likelihood whose minimiser is \cref{eq:pi}, in $D$ a strictly convex quadratic (above) and in $\sigma^2$ a separable problem whose stationary point is the stated variance. Each block update therefore does not increase $J$, which is bounded below, so $J$ converges. With $\lambda_L>0$ the term $-\frac{\lambda_L}{2m_s}z^{\top}W_s z$ is handled by the linear bound $-z^{\top}W_s z\le -2 z^{\top}W_s z^{(t)} + z^{(t)\top}W_s z^{(t)}$, valid when $W_s$ is positive semi-definite and used as a majoriser otherwise, as in~\cite{ziko2020laplacianshot,zanella2024transclip}. \qed

\section{Online algorithm}
\begin{algorithm}[h]
\caption{\method, online inference on a stream}
\label{alg:online}
\begin{algorithmic}[1]
\Require anchors $T$, graph $L_T$, forgetting $\varrho\in(0,1]$; state $(n,S,q,N)\gets 0$, $D\gets 0$, $\pi\gets\mathbf{1}/K$
\For{each batch $X^{(b)}$}
  \State $\hat{y}\gets\mathrm{softmax}(\tau X^{(b)}T^{\top})$; E-step with current $(D,\pi,\sigma^2)$ \Cref{eq:estep}
  \State $(n,S,q,N)\gets\varrho\,(n,S,q,N) + \big(z^{\top}\mathbf{1},\, z^{\top}X^{(b)},\, \textstyle\sum_i x_i^{\odot2},\, |X^{(b)}|\big)$
  \State M-steps of \cref{alg:batch} from $(n,S,q,N)$; re-run the E-step on $X^{(b)}$ (two rounds)
  \State \Return posterior $z$ of $X^{(b)}$
\EndFor
\end{algorithmic}
\end{algorithm}

\section{Empirical Bayes details}
With $P$ the present classes, the marginal covariance of $e_{:j,P}$ in pseudo-count units is $\tilde{Q}_P^{-1} + N_{z,P}^{-1}$ where $\tilde{Q}_P$ is the Schur complement of $Q_\lambda$ onto $P$, because marginalising a GMRF onto a subset of nodes replaces the precision by its Schur complement~\cite{rue2005gmrf}. The evidence of \cref{eq:evidence} sums the $d$ independent dimensions after whitening $E$ by $\sqrt{\kappa\bar\sigma^2}$. One Cholesky factorisation of a $|P|\times|P|$ matrix per grid point suffices; the Schur complement requires one solve with $Q_{\lambda,AA}$, which is sparse and well conditioned when $\alpha>0$. The rule is disabled when $|P|<3$, since the evidence is then nearly flat in $\lambda$.

The batch sampler draws $K_{\mathrm{eff}}$ uniformly in $[k_{\min},k_{\max}]$, selects $K_{\mathrm{eff}}$ classes without replacement, pools their test images and draws $B$ of them without replacement (all of them if fewer than $B$ are available). The online sampler draws, for each class, Dirichlet($\gamma$) proportions over $K$ slots, assigns the class's shuffled images to slots accordingly, shuffles within slots and concatenates the slots; $\gamma=-1$ denotes the separate stream. These re-implementations follow the description of~\cite{zanella2025stata}; the released adapter also runs \method inside the official harness so that the exact official samplers can be used.
